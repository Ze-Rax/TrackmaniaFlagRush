// MARK: Lifecycle

Void BeforeYield() {
	foreach (Event in Playloop::GetPendingEvents()) {
		switch (Event.Type) {
			case Playloop::C_EventType_FlagReset,
					Playloop::C_EventType_PlayerPickedUpFlag,
					Playloop::C_EventType_PlayerPassedFlag,
					Playloop::C_EventType_PlayerDroppedFlag: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerRequestedRespawnButIsTooFast: {
				declare CSmPlayer Player = Playloop::GetPlayer(Event.Player);
				if (Player != Null) {
					UIModule_RespawnFlash::Flash(Player);
				}
			}
		}
	}

	foreach (Event in Progression::GetPendingEvents()) {
		switch (Event.Type) {
			case Progression::C_EventType_SegmentStart: {
				switch (Event.Segment) {
					case Progression::C_Segment_Map: {
						UIModule_MapMarkers::Update();
					}
					case Progression::C_Segment_WarmUp: {
						UIModule_MapMarkers::ShowFlagSpawnMarkers();
					}
					case Progression::C_Segment_Podium: {
						UIModule_MapMarkers::Hide();
						declare CSmScore[] PodiumPlayerScores;
						declare Text[] PodiumPlayerLogins;
						for (I, 0, 2) {
							if (!Scores.existskey(I)) {
								break;
							}
							PodiumPlayerScores.add(Scores[I]);
							PodiumPlayerLogins.add(Scores[I].User.Login);
						}
						UIManager.UIAll.UISequence_PodiumPlayersWin = TL::Join(",", PodiumPlayerLogins);
						UIModule_PodiumBackground::SetContent(PodiumPlayerScores);
						Layers::Attach(UIModule_PodiumBackground::C_LayerName);
						UIModule_PodiumStatistics::Update();
						Layers::Attach(UIModule_PodiumStatistics::C_LayerName);
						Layers::Detach(UIModule_ScoresHeader::C_LayerName);
					}
				}
			}
			case Progression::C_EventType_SegmentEnd: {
				switch (Event.Segment) {
					case Progression::C_Segment_Turn: {
						UIModule_FlagMarker::Update();
					}
					case Progression::C_Segment_WarmUp: {
						UIModule_MapMarkers::HideFlagSpawnMarkers();
					}
					case Progression::C_Segment_Podium: {
						UIModule_MapMarkers::Show();
						Layers::Detach(UIModule_PodiumBackground::C_LayerName);
						Layers::Detach(UIModule_PodiumStatistics::C_LayerName);
						Layers::Attach(UIModule_ScoresHeader::C_LayerName);
					}
				}
			}
		}
	}

	// React to events
	foreach (Event in Settings::GetPendingEvents()) {
		switch (Event.Name) {
			case SettingNames::S_ModeCommands_AdminLoginsCsv: {
				UIModule_ModeCommands::RemoveAllUserPermissions();
				declare Text[] ModeCommandsAdminLogins = TL::Split(",", Event.NewValue.ValueText);
				foreach(Login in ModeCommandsAdminLogins) {
					UIModule_ModeCommands::SetSuperAdmin(Login);
				}
			}
			case SettingNames::S_NbMapsToWinMatch: {
				SharedNetVars::SetNbMapsToWinMatch(Event.NewValue.ValueInteger);
			}
			case SettingNames::S_NbRoundsToWinMap: {
				SharedNetVars::SetNbRoundsToWinMap(Event.NewValue.ValueInteger);
			}
			case SettingNames::S_NbFlagsToWinRound: {
				SharedNetVars::SetNbFlagsToWinRound(Event.NewValue.ValueInteger);
			}
		}
	}

	declare Boolean UpdateRadar = False;
	foreach (Event in PlayerLifecycleEvents::GetPendingEvents()) {
		switch (Event.Type) {
			case PlayerLifecycleEvents::C_EventType_PlayerAdded: {
				if (Players.existskey(Event.Player.PlayerId)) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerRemoved: {
				if (!Event.Player.Previously.WasSpectator) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerToSpectator, PlayerLifecycleEvents::C_EventType_SpectatorToPlayer: {
				UpdateRadar = True;
			}
		}
	}

	if (UpdateRadar) {
		Layers::Update(UIModule_Radar::C_LayerName, UIModule_Radar::GetManialink());
	}

	foreach (Event in EnabledVehicles::GetPendingEvents()) {
		UIModule_VehicleSelection::SetEnabledVehicles(Event.EnabledVehicles);
	}
}

Void Yield() {
	UIModule_ModeCommands::Yield();
	foreach (Command in UIModule_ModeCommands::GetPendingCommands()) {
		Commands::Publish(Command);
	}

	UIModule_VehicleSelection::Yield();
	foreach (Event in UIModule_VehicleSelection::GetPendingEvents()) {
		Players::SetSelectedVehicle(Event.Player, Event.SelectedVehicle);
	}
}

Void Unload() {
	Private_DestroyModuleLayers();
	Private_UnloadModules();
}

Void Load() {
	Unload();
	Private_LoadModules();
}

Void LoadHud() {
	Private_CreateModuleLayers();
}
