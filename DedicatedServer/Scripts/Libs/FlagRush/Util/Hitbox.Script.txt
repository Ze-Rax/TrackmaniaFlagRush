#Const Version					"2021-12-22"
#Const ScriptName				"Hitbox.Script.txt"

#Include "MathLib" as ML
#Include "Libs/FlagRush/CommonLibs/OBB.Script.txt" as OBB

#Const C_FlagColliderRadius 1.

#Struct K_OBBTransform {
	Vec3 Extents; //left up forward
	Vec3 Offsets; //0 up forward
}

Real Private_MinComponent(Vec3 _Vec) {
	return ML::Min(_Vec.X, ML::Min(_Vec.Y, _Vec.Z));
}

Vec3[] Private_ComputeBasis(CSmPlayer _Player) {
	return [
		_Player.LeftDirection,
		_Player.UpDirection,
		ML::CrossProduct(_Player.LeftDirection, _Player.UpDirection)
	];
}

OBB::K_OBB Private_ComputeOBB(CSmPlayer _Player, K_OBBTransform _ObbTransform) {
	declare Vec3[] Basis = Private_ComputeBasis(_Player);
	// Ignore Offset.X as we assume that the bounding box is center on the X (left right) axis
	declare Vec3 Center = _Player.Position + Basis[1] * _ObbTransform.Offsets.Y + Basis[2] * _ObbTransform.Offsets.Z;

	return OBB::K_OBB {
		C = Center,
		M = Basis,
		HW = _ObbTransform.Extents
	};
}

/**
 * Checks if two players collide
 */
Boolean AreColliding(CSmPlayer _PlayerA, K_OBBTransform _ObbTransformA, CSmPlayer _PlayerB, K_OBBTransform _ObbTransformB) {
	declare Real MaxDistance = ML::Length(_ObbTransformA.Extents + _ObbTransformB.Extents);
	//if both players are more than MaxDistance away from each other there is no case where they do collide
	if (ML::Distance(_PlayerA.Position, _PlayerB.Position)  > MaxDistance) return False;

	declare Real MinDistance = Private_MinComponent(_ObbTransformA.Extents) + Private_MinComponent(_ObbTransformB.Extents);
	//if both players are less than MinDistance away from each other there is no case where they do NOT collide
	if (ML::Distance(_PlayerA.Position, _PlayerB.Position)  < MinDistance) return True;

	declare OBB::K_OBB ObbA = Private_ComputeOBB(_PlayerA, _ObbTransformA);
	declare OBB::K_OBB ObbB = Private_ComputeOBB(_PlayerB, _ObbTransformB);
	return OBB::TestOBBOBB(ObbA,ObbB);
}

/**
 * Checks if a player and the flag collide
 */
Boolean AreColliding(CSmPlayer _Player, K_OBBTransform _ObbTransform, Vec3 _FlagPos) {


	if (ML::Distance(_Player.Position, _FlagPos)  < 2 * C_FlagColliderRadius) return True;


	return False;
}
