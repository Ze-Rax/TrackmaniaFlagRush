#Const C_PlayerLifecycle_PlayerAdded "PlayerAdded"
#Const C_PlayerLifecycle_PlayerRemoved "PlayerRemoved" ///< Note: Player in event will be null, as the player already left and the instance was removed
#Const C_PlayerLifecycle_SpectatorToPlayer "SpectatorToPlayer"
#Const C_PlayerLifecycle_PlayerToSpectator "PlayerToSpectator"
#Const C_PlayerLifecycle_PlayerClanChangeRequest "PlayerClanChangeRequest"
#Const C_PlayerLifecycle_PlayerClanChange "PlayerClanChange"

#Struct K_Event {
	Text Type;
	CSmPlayer Player;
	CUser User;
	Boolean IsSpectator;
	Integer Clan;
	Integer PreviousClan; ///< For Multiclan usage
}

declare K_Event[] G_PendingEvents;

/**
 * Get all pending events. Pending events are flushed and collected by PlayerLifecycleEvents::Yield().
 */
K_Event[] GetPendingEvents() {
	return G_PendingEvents;
}

/**
 * Flushes the old event arrays and collects new events.
 * Should be called in ***MB_Yield***.
 */
Void Yield() {
	G_PendingEvents.clear();

	// Check default mode events for join and leave
	foreach (Event in PendingEvents) {
		switch (Event.Type) {
			case CSmModeEvent::EType::OnPlayerAdded: {
				// Initilize player
				declare CUser User = Event.Player.User; // Event.User is Null
				declare Boolean PlayerLifecycleEvents_WasSpectator for User = False;
				declare Integer PlayerLifecycleEvents_PrevRequestedClan for User = 0;
				declare Integer PlayerLifecycleEvents_PrevClan for User = 0;
				PlayerLifecycleEvents_WasSpectator = Spectators.existskey(Event.Player.Id); // Do not set in one line as it could be reconnect and user was already present
				PlayerLifecycleEvents_PrevRequestedClan = Event.Player.RequestedClan;
				PlayerLifecycleEvents_PrevClan = Event.Player.CurrentClan;

				G_PendingEvents.add(K_Event{
					Type = C_PlayerLifecycle_PlayerAdded,
					Player = Event.Player,
					User = User,
					IsSpectator = PlayerLifecycleEvents_WasSpectator,
					Clan = Event.Player.CurrentClan,
					PreviousClan = -1 // Invalid value; He just joined
				});
			}
			case CSmModeEvent::EType::OnPlayerRemoved: {
				declare Boolean PlayerLifecycleEvents_WasSpectator for Event.User = False;
				declare Integer PlayerLifecycleEvents_PrevClan for Event.User = 0;
				G_PendingEvents.add(K_Event{
					Type = C_PlayerLifecycle_PlayerRemoved,
					Player = Event.Player,
					User = Event.User,
					IsSpectator = PlayerLifecycleEvents_WasSpectator,
					Clan = PlayerLifecycleEvents_PrevClan,
					PreviousClan = PlayerLifecycleEvents_PrevClan
				});
			}
		}
	}

	foreach (Player in Players) {
		declare Boolean PlayerLifecycleEvents_WasSpectator for Player.User = False;
		declare Integer PlayerLifecycleEvents_PrevRequestedClan for Player.User = 0;
		declare Integer PlayerLifecycleEvents_PrevClan for Player.User = 0;

		// Check spectators that switched to player
		if (PlayerLifecycleEvents_WasSpectator) {
			PlayerLifecycleEvents_WasSpectator = False;
			G_PendingEvents.add(K_Event{
				Type = C_PlayerLifecycle_SpectatorToPlayer,
				Player = Player,
				User = Player.User,
				IsSpectator = False,
				Clan = Player.CurrentClan,
				PreviousClan = PlayerLifecycleEvents_PrevClan
			});
		}

		// Check clan change requests
		if (PlayerLifecycleEvents_PrevRequestedClan != Player.RequestedClan) {
			G_PendingEvents.add(K_Event{
				Type = C_PlayerLifecycle_PlayerClanChangeRequest,
				Player = Player,
				User = Player.User,
				IsSpectator = False,
				Clan = Player.RequestedClan,
				PreviousClan = PlayerLifecycleEvents_PrevRequestedClan
			});
			// Update after event creation to provide the previous value as well
			PlayerLifecycleEvents_PrevRequestedClan = Player.RequestedClan;
		}

		// Check actual clan changes
		if (PlayerLifecycleEvents_PrevClan != Player.CurrentClan) {
			G_PendingEvents.add(K_Event{
				Type = C_PlayerLifecycle_PlayerClanChange,
				Player = Player,
				User = Player.User,
				IsSpectator = False,
				Clan = Player.CurrentClan,
				PreviousClan = PlayerLifecycleEvents_PrevClan
			});
			// Update after event creation to provide the previous value as well
			PlayerLifecycleEvents_PrevClan = Player.CurrentClan;
		}
	}

	// Check players that switched to spectator
	foreach (Spectator in Spectators) {
		declare Boolean PlayerLifecycleEvents_WasSpectator for Spectator.User = False;
		declare Integer PlayerLifecycleEvents_PrevClan for Spectator.User = 0;

		if (!PlayerLifecycleEvents_WasSpectator) {
			PlayerLifecycleEvents_WasSpectator = True;
			G_PendingEvents.add(K_Event{
				Type = C_PlayerLifecycle_PlayerToSpectator,
				Player = Spectator,
				User = Spectator.User,
				IsSpectator = True,
				Clan = Spectator.CurrentClan, // Should be 0 for spectators
				PreviousClan = PlayerLifecycleEvents_PrevClan
			});
		}
	}
}

Void Unload() {
	G_PendingEvents.clear();
}

Void Load() {
	Unload();
}
