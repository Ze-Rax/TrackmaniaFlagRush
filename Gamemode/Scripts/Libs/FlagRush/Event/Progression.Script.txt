/**
 * Events for the general progression of the FlagRush mode.
 *
 * Provides events for begin and end of different segments, like warmup, maps, matches, etc.
 * Consuming scripts should only access the pending using the GetPendingEvents function.
 * Events are published from the respective mode using the Mode_PublishEvent function. They should not be used by consuming scripts.
 * This script therefore only serves as a point that is accessible to libraries, as they cannot access members from the main mode file.
 *
 * #RequireContext CSmMode
 */

#Const C_EventType_SegmentStart 1
#Const C_EventType_SegmentEnd 2
#Const C_EventType_SegmentInvalid 3
#Const C_EventType_EndTimeChanged 4

#Const C_Segment_Server 1
#Const C_Segment_Match 2
#Const C_Segment_Map 3
#Const C_Segment_Round 4
#Const C_Segment_Turn 5
#Const C_Segment_Overtime 6
#Const C_Segment_WarmUp 7
#Const C_Segment_Pause 8
#Const C_Segment_Podium 9

#Const C_EventTypeNames [
	1 => "SegmentStart",
	2 => "SegmentEnd",
	3 => "SegmentInvalid",
	4 => "EndTimeChanged"
]

#Const C_SegmentNames [
  1 => "Server",
  2 => "Match",
  3 => "Map",
  4 => "Round",
  5 => "Turn",
  6 => "Overtime",
  7 => "WarmUp",
  8 => "Pause",
  9 => "Podium"
]

#Struct K_Event {
  Integer Type;
  Integer Segment;
}

declare Boolean G_ServerIsRunning;
declare Boolean G_MatchIsRunning;
declare Boolean G_MapIsRunning;
declare Boolean G_RoundIsRunning;
declare Boolean G_TurnIsRunning;
declare Boolean G_OvertimeIsRunning;
declare Boolean G_WarmUpIsRunning;
declare Boolean G_PauseIsRunning;
declare Boolean G_PodiumIsRunning;

declare Boolean G_MatchIsValid;
declare Boolean G_MapIsValid;
declare Boolean G_RoundIsValid;

declare Boolean G_MatchEndRequested;
declare Boolean G_RoundPauseRequested;
declare Boolean G_RoundEndRequested;

// Need a separate buffer for events before publishing.
// If events are added to pending events directly, then during yield they would be cleared before other libraries could see them.
// This way it is guaranteed that they are visible for exactly one yield, no matter if other libraries yield before or after this library.
declare K_Event[] G_EventsAggregationBuffer;
declare K_Event[] G_PendingEvents;

// MARK: API

K_Event[] GetPendingEvents() {
  return G_PendingEvents;
}

// Segment running states

Boolean ServerIsRunning() {
  return G_ServerIsRunning;
}

Boolean MatchIsRunning() {
  return G_MatchIsRunning;
}

Boolean MapIsRunning() {
  return G_MapIsRunning;
}

Boolean RoundIsRunning() {
  return G_RoundIsRunning;
}

Boolean TurnIsRunning() {
  return G_TurnIsRunning;
}

Boolean OvertimeIsRunning() {
  return G_OvertimeIsRunning;
}

Boolean WarmUpIsRunning() {
  return G_WarmUpIsRunning;
}

Boolean PauseIsRunning() {
  return G_PauseIsRunning;
}

Boolean PodiumIsRunning() {
  return G_PodiumIsRunning;
}

Boolean SegmentIsRunning(Integer _Segment) {
  switch(_Segment) {
    case C_Segment_Server: return ServerIsRunning();
    case C_Segment_Match: return MatchIsRunning();
    case C_Segment_Map: return MapIsRunning();
    case C_Segment_Round: return RoundIsRunning();
    case C_Segment_Turn: return TurnIsRunning();
    case C_Segment_Overtime: return OvertimeIsRunning();
    case C_Segment_WarmUp: return WarmUpIsRunning();
    case C_Segment_Pause: return PauseIsRunning();
    case C_Segment_Podium: return PodiumIsRunning();
  }
  return False;
}

// Segment end requests

Boolean MatchEndRequested() {
  return G_MatchEndRequested;
}

Void RequestMatchEnd() {
  G_MatchEndRequested = True;
}

Boolean MapEndRequsted() {
  // Contradictory to its name, MatchEndRequested is generally used to skip a map, not the whole match.
  return MatchEndRequested() || MatchEndRequested;
}

Void RequestMapEnd() {
  MatchEndRequested = True;
}

Boolean RoundEndRequested() {
  return MapEndRequsted() || G_RoundEndRequested;
}

Void RequestRoundEnd() {
  G_RoundEndRequested = True;
}

Boolean RoundPauseRequested() {
  return G_RoundPauseRequested;
}

Void RequestRoundPause() {
  G_RoundPauseRequested = True;
}

// Segment validities

Boolean MatchIsValid() {
  return G_MatchIsValid;
}

Void InvalidateMatch() {
  G_MatchIsValid = False;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentInvalid,
    Segment = C_Segment_Match
  });
}

Boolean MapIsValid() {
  return MatchIsValid() && G_MapIsValid && !MapEndRequsted();
}

Void InvalidateMap() {
  G_MapIsValid = False;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentInvalid,
    Segment = C_Segment_Map
  });
}

Boolean RoundIsValid() {
  return MapIsValid() && G_RoundIsValid && !RoundEndRequested();
}

Void InvalidateRound() {
  G_RoundIsValid = False;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentInvalid,
    Segment = C_Segment_Round
  });
}

// EndTime

Integer GetEndTime() {
  return EndTime;
}

Void SetEndTime(Integer _EndTime) {
  EndTime = _EndTime;
  declare netwrite Integer Net_FlagRush_Progression_EndTime for Teams[0] = -1;
	Net_FlagRush_Progression_EndTime = EndTime;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_EndTimeChanged
  });
}

Void StartServer() {
  G_ServerIsRunning = True;
  declare netwrite Boolean Net_FlagRush_Progression_ServerIsRunning for Teams[0] = True;
  Net_FlagRush_Progression_ServerIsRunning = True;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentStart,
    Segment = C_Segment_Server
  });
}

  G_ServerIsRunning = False;
  declare netwrite Boolean Net_FlagRush_Progression_ServerIsRunning for Teams[0] = False;
  Net_FlagRush_Progression_ServerIsRunning = False;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentEnd,
    Segment = C_Segment_Server
  });
}

Void EndRound() {
  if (G_OvertimeIsRunning) {
    G_OvertimeIsRunning = False;
    declare netwrite Boolean Net_FlagRush_Progression_OvertimeIsRunning for Teams[0] = False;
    Net_FlagRush_Progression_OvertimeIsRunning = False;
    G_EventsAggregationBuffer.add(K_Event{
      Type = C_EventType_SegmentEnd,
      Segment = C_Segment_Overtime
    });
  }

  G_RoundIsRunning = False;
  declare netwrite Boolean Net_FlagRush_Progression_RoundIsRunning for Teams[0] = False;
  Net_FlagRush_Progression_RoundIsRunning = False;
  G_EventsAggregationBuffer.add(K_Event{
    Type = C_EventType_SegmentEnd,
    Segment = C_Segment_Round
  });
}
