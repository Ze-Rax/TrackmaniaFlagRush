// MARK: Helper Functions

Void Private_EnableTimeLimitWarning() {
	G_TimeLimitWarningsEnabled.clear();
	foreach (WarningSeconds in C_TimeLimitWarning_Seconds) {
		G_TimeLimitWarningsEnabled[WarningSeconds] = Settings::GetInteger(SettingNames::S_RoundTimeLimitSeconds) > WarningSeconds;
	}
}

Void Private_ClearAllMessages() {
	UIManager.UIAll.ClearMessages();
	foreach (Player in AllPlayers) {
		declare CUIConfig UI <=> UIManager.GetUI(Player);
		if (UI != Null) {
			UI.ClearMessages();
		}
	}
}

Text Private_GetTeamColorCode(Integer _Clan) {
	return """${{{ CL::RgbToHex3(Teams::GetMidColor(_Clan)) }}}""";
}

Text Private_GetColoredTeamName(Integer _Clan) {
	return """$<${{{ CL::RgbToHex3(Teams::GetMidColor(_Clan)) }}}{{{ Teams[_Clan-1].Name }}}$>""";
}

Text Private_GetColoredPlayerName(CSmPlayer _Player) {
	return """$<{{{ Private_GetTeamColorCode(_Player.CurrentClan) }}}{{{ _Player.User.Name }}}$>""";
}

Void Private_QueueBigMessage(Text _Message, Integer _Duration, Integer _Priority, CUIConfig::EUISound _Sound) {
	// Private_ClearAllMessages();
	// UIManager.UIAll.QueueMessage(_Duration, _Priority, CUIConfig::EMessageDisplay::Big, _Message, _Sound, 0);
	// Using Message library to make it possible to replace big messages instantly (for example when flag changes carrier often in a short timeframe)
	Message::CleanBigMessages();
	Message::SendBigMessage(_Message, _Duration, _Priority, _Sound, 0);
}

Void BeforeYield() {
	foreach (Event in Playloop::GetPendingEvents()) {
		switch (Event.Type) {
			case Playloop::C_EventType_FlagReset: {
				if (!Event.Silent) {
					OnFlagReset();
				}
			}
			case Playloop::C_EventType_PlayerPickedUpFlag: {
				OnFlagPickedUp(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_PlayerPassedFlag: {
				// Note that the event is from the perspective of the old carrier who passed the flag
				OnFlagPickedUp(Playloop::GetPlayer(Event.RelatedPlayer));
			}
			case Playloop::C_EventType_PlayerDroppedFlag: {
				OnFlagDropped(Playloop::GetScore(Event.Player));
			}
			case Playloop::C_EventType_PlayerScoredFlag: {
				OnFlagScored(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_FlagCarrierTeleported: {
				OnFlagCarrierTeleported();
			}
			case Playloop::C_EventType_PlayerOutOfBounds: {
				OnPlayerOutOfBounds(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_PlayerRespawned: {
				OnPlayerRespawned(Playloop::GetPlayer(Event.Player));
			}
		}
	}
}

Void Yield() {
	Message::Yield();

	foreach (Event in PlayerLifecycleEvents::GetPendingEvents()) {
		if (Event.Type == PlayerLifecycleEvents::C_EventType_PlayerAdded) {
			declare CSmPlayer Player <=> AllPlayers.get(Event.Player.PlayerId, Null);
			if (Player != Null) {
				Welcome(Player);
			}
		}
	}

	// Time warnings
	if (Progression::TurnIsRunning()) {
		foreach (WarningSeconds => Enabled in G_TimeLimitWarningsEnabled) {
			if (!Enabled || (EndTime - Now) / 1000 > WarningSeconds) {
				continue;
			}
			G_TimeLimitWarningsEnabled[WarningSeconds] = False;
			TimeLimitSecondsRemaining(WarningSeconds, Durations::C_EventMessage);
		}
	}

	foreach (Event in Progression::GetPendingEvents()) {
		switch (Event.Type) {
			// Segment Start
			case Progression::C_EventType_SegmentStart: {
				switch (Event.Segment) {
					case Progression::C_Segment_Round: {
						OnRoundStarted(Progression::GetRoundNumber());
						Private_EnableTimeLimitWarning();
					}
					case Progression::C_Segment_Overtime: {
						OnOvertimeStarted();
					}
					case Progression::C_Segment_MatchEndScoresTable: {
						if (Event.IsInvalid) {
							OnMatchSkipped();
							break;
						}
						declare Integer WinnerClan = Scores::EvaluateRound();
						if (WinnerClan == 0) {
							OnMatchDraw();
						} else {
							OnMatchWon(WinnerClan);
						}
					}
					case Progression::C_Segment_WarmUp: {
						OnWarmUpStarted();
						if (Settings::GetBoolean(SettingNames::S_WarmUpWaitForApproval)) {
							OnWarmUpWaitForApproval();
						}
					}
					case Progression::C_Segment_Pause: {
						OnPauseStarted();
					}
				}
			}
			// Segment End
			case Progression::C_EventType_SegmentEnd: {
				switch (Event.Segment) {
					case Progression::C_Segment_Map: {
						if (Event.IsInvalid) {
							OnMapSkipped();
							break;
						}
						declare Integer WinnerClan = Scores::EvaluateMap();
						if (WinnerClan == 0) {
							OnMapDraw();
						} else {
							OnMapWon(WinnerClan);
						}
					}
					case Progression::C_Segment_Round: {
						if (Event.IsInvalid) {
							OnRoundSkipped();
							break;
						}
						declare Integer WinnerClan = Scores::EvaluateRound();
						if (WinnerClan == 0) {
							OnRoundDraw();
						} else {
							OnRoundWon(WinnerClan);
						}
					}
					case Progression::C_Segment_WarmUp: {
						OnWarmUpEnded();
					}
					case Progression::C_Segment_Pause: {
						OnPauseEnded();
					}
				}
			}
			// Segment Invalid
			case Progression::C_EventType_SegmentInvalid: {
				switch (Event.Segment) {
					case Progression::C_Segment_Map: {
						OnInvalidMap();
					}
				}
			}
			// Time limit reached
			case Progression::C_EventType_TimeLimitReached: {
				OnTimeLimitReached();
			}
			// Warm Up Approval changed
			case Progression::C_EventType_WarmUpApprovalChanged: {
				if (WarmUpReadyUp::IsWarmUpEndApproved()) {
					OnWarmUpEndApproved();
				} else {
					OnWarmUpWaitForApproval();
				}
			}
			//  Match skipped due to no players being left
			case Progression::C_EventType_MatchSkippedDueToNoPlayersLeft: {
				OnMatchSkippedDueToNoPlayersLeft();
			}
			// Restart due to critical mode setting changes
			case Progression::C_EventType_RestartTurnDueToCriticalModeSettingChanged: {
				OnTurnRestartedDueToCriticalSettingChanged();
			}
		}
	}
}
