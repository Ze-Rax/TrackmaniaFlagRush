Void Private_ClearAllMessages() {
	UIManager.UIAll.ClearMessages();
	foreach (Player in AllPlayers) {
		declare CUIConfig UI <=> UIManager.GetUI(Player);
		if (UI != Null) {
			UI.ClearMessages();
		}
	}
}

Text Private_GetTeamColorCode(Integer _Clan) {
	return """${{{ CL::RgbToHex3(Teams::GetMidColor(_Clan)) }}}""";
}

Text Private_GetColoredTeamName(Integer _Clan) {
	return """$<${{{ CL::RgbToHex3(Teams::GetMidColor(_Clan)) }}}{{{ Teams[_Clan-1].Name }}}$>""";
}

Text Private_GetColoredPlayerName(CSmPlayer _Player) {
	return """$<{{{ Private_GetTeamColorCode(_Player.CurrentClan) }}}{{{ _Player.User.Name }}}$>""";
}

Void Private_QueueBigMessage(Text _Message, Integer _Duration, Integer _Priority, CUIConfig::EUISound _Sound) {
	// Private_ClearAllMessages();
	// UIManager.UIAll.QueueMessage(_Duration, _Priority, CUIConfig::EMessageDisplay::Big, _Message, _Sound, 0);
	// Using Message library to make it possible to replace big messages instantly (for example when flag changes carrier often in a short timeframe)
	Message::CleanBigMessages();
	Message::SendBigMessage(_Message, _Duration, _Priority, _Sound, 0);
}

Void OnFlagPickedUp(CSmPlayer _Player) {
	declare Text Message = """{{{ Private_GetColoredPlayerName(_Player) }}} has the $<{{{ Private_GetTeamColorCode(_Player.CurrentClan) }}}flag$>""";
	declare Vec3 TeamColor = Teams::GetMidColor(_Player.CurrentClan);
	Private_QueueBigMessage(Message, Durations::C_Duration_EventMessage, C_MessagePriority_Event, CUIConfig::EUISound::Checkpoint);
	UIModule_EventFeed::SendMessage(Message, UIModule_EventFeed::C_IconName_Flag, TeamColor);
}

Void OnFlagScored(CSmPlayer _Scorer) {
	declare Text Message = """{{{ Private_GetColoredPlayerName(_Scorer) }}} scored for {{{ Private_GetColoredTeamName(_Scorer.CurrentClan) }}}""";
	declare Vec3 TeamColor = Teams::GetMidColor(_Scorer.CurrentClan);

	UIModule_EventFeed::SendMessage(Message, UIModule_EventFeed::C_IconName_Flag, TeamColor);
}

	UIModule_EventFeed::SendMessage(Message, UIModule_EventFeed::C_IconName_Error);

Void BeforeYield() {
	foreach (Event in Playloop::GetPendingEvents()) {
		switch (Event.Type) {
			case Playloop::C_EventType_FlagReset: {
				if (!Event.Silent) {
					OnFlagReset();
				}
			}
			case Playloop::C_EventType_PlayerPickedUpFlag: {
				OnFlagPickedUp(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_PlayerPassedFlag: {
				// Note that the event is from the perspective of the old carrier who passed the flag
				OnFlagPickedUp(Playloop::GetPlayer(Event.RelatedPlayer));
			}
			case Playloop::C_EventType_PlayerDroppedFlag: {
				OnFlagDropped(Playloop::GetScore(Event.Player));
			}
			case Playloop::C_EventType_PlayerScoredFlag: {
				OnFlagScored(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_FlagCarrierTeleported: {
				OnFlagCarrierTeleported();
			}
			case Playloop::C_EventType_PlayerOutOfBounds: {
				OnPlayerOutOfBounds(Playloop::GetPlayer(Event.Player));
			}
			case Playloop::C_EventType_PlayerRespawned: {
				OnPlayerRespawned(Playloop::GetPlayer(Event.Player));
			}
		}
	}
}
