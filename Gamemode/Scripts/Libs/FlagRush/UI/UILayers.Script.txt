/**
 * Sends relevant progression mode settings values to the clients via netwrite.
 */
Void Net_SendModeScriptProgressionSettings(Integer NbMapsToWinMatch, Integer NbRoundsToWinMap, Integer NbFlagsToWinRound) {
	declare netwrite Integer Net_FlagRush_NbMapsToWinMatch for Teams[0];
	Net_FlagRush_NbMapsToWinMatch = NbMapsToWinMatch;

	declare netwrite Integer Net_FlagRush_NbRoundsToWinMap for Teams[0];
	Net_FlagRush_NbRoundsToWinMap = NbRoundsToWinMap;

	declare netwrite Integer Net_FlagRush_NbFlagsToWinRound for Teams[0];
	Net_FlagRush_NbFlagsToWinRound = NbFlagsToWinRound;
}

/* Lifecycle */

Void BeforeYield() {
	foreach (Event in Playloop::GetPendingEvents()) {
		switch (Event.Type) {
			case Playloop::C_EventType_FlagReset: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerPickedUpFlag: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerPassedFlag: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerDroppedFlag: {
				UIModule_FlagMarker::Update();
			}
		}
	}
}

Void Yield() {
	UIModule_ModeCommands::Yield();

	foreach (Event in Settings::GetPendingEvents()) {
		switch (Event.Name) {
			case SettingNames::S_ModeCommands_AdminLoginsCsv: {
				UIModule_ModeCommands::RemoveAllUserPermissions();
				declare Text[] ModeCommandsAdminLogins = TL::Split(",", Event.NewValue.ValueText);
				foreach(Login in ModeCommandsAdminLogins) {
					UIModule_ModeCommands::SetSuperAdmin(Login);
				}
			}
		}
	}

	declare Boolean UpdateRadar = False;
	foreach (Event in PlayerLifecycleEvents::GetPendingEvents()) {
		switch (Event.Type) {
			case PlayerLifecycleEvents::C_EventType_PlayerAdded: {
				if (Players.existskey(Event.Player.PlayerId)) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerRemoved: {
				if (!Event.Player.Previously.WasSpectator) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerToSpectator, PlayerLifecycleEvents::C_EventType_SpectatorToPlayer: {
				UpdateRadar = True;
			}
		}
	}

	if (UpdateRadar) {
		Layers::Update(C_LayerName_Radar, UIModule_Radar::GetManialink());
	}

	foreach (Event in Progression::GetPendingEvents()) {
		if (Event.Type == Progression::C_EventType_SegmentStart && Event.Segment == Progression::C_Segment_Map) {
			UIModule_MapMarkers::Update();
		} else if (Event.Type == Progression::C_EventType_SegmentStart && Event.Segment == Progression::C_Segment_WarmUp) {
			UIModule_MapMarkers::ShowFlagSpawnMarkers();
		} else if (Event.Type == Progression::C_EventType_SegmentEnd && Event.Segment == Progression::C_Segment_WarmUp) {
			UIModule_MapMarkers::HideFlagSpawnMarkers();
		} else if (Event.Type == Progression::C_EventType_SegmentStart && Event.Segment == Progression::C_Segment_Podium) {
			UIModule_MapMarkers::Hide();
		} else if (Event.Type == Progression::C_EventType_SegmentEnd && Event.Segment == Progression::C_Segment_Podium) {
			UIModule_MapMarkers::Show();
		} else if (Event.Type == Progression::C_EventType_SegmentEnd && Event.Segment == Progression::C_Segment_Turn) {
			UIModule_FlagMarker::Update();
		}
	}

	foreach (Command in UIModule_ModeCommands::GetPendingCommands()) {
		Commands::Publish(Command);
	}
}

	Private_DestroyModuleLayers();
	Private_UnloadModules();

	Private_LoadModules();
