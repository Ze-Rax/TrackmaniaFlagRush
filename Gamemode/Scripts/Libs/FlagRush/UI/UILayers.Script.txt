Void Private_CreateModuleLayers() {

	Layers::Attach(UIModule_EventFeed::C_LayerName);

	// Flag Marker
	Layers::Create(UIModule_FlagMarker::C_LayerName, UIModule_FlagMarker::GetManialink());
	Layers::SetType(UIModule_FlagMarker::C_LayerName, UIModule_FlagMarker::C_LayerType);
	Layers::Attach(UIModule_FlagMarker::C_LayerName);

	// Flag Possession Indicator
	Layers::Create(UIModule_FlagPossessionIndicator::C_LayerName, UIModule_FlagPossessionIndicator::GetManialink());
	Layers::SetType(UIModule_FlagPossessionIndicator::C_LayerName, UIModule_FlagPossessionIndicator::C_LayerType);
	Layers::Attach(UIModule_FlagPossessionIndicator::C_LayerName);

	// Scores header
	Layers::Create(C_LayerName_ScoresHeader, UIModule_ScoresHeader::GetManialink());
	Layers::SetType(C_LayerName_ScoresHeader, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_ScoresHeader);

	// Scores table
	Layers::Create(C_LayerName_ScoresTable, UIModule_ScoresTable::GetManialink());
	Layers::SetType(C_LayerName_ScoresTable, CUILayer::EUILayerType::ScoresTable);
	Layers::Attach(C_LayerName_ScoresTable);

	// Sound
	Layers::Create(C_LayerName_Sound, UIModule_Sounds::GetManialink());
	Layers::SetType(C_LayerName_Sound, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_Sound);

	// Menu
	Layers::Create(C_LayerName_Menu, UIModule_Menu::GetManialink());
	Layers::SetType(C_LayerName_Menu, CUILayer::EUILayerType::InGameMenu);
	Layers::Attach(C_LayerName_Menu);


	// Respawn Bar
	Layers::Create(C_LayerName_Respawn, UIModule_RespawnTimer::GetManialink());
	Layers::SetType(C_LayerName_Respawn, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_Respawn);

	// Radar
	Layers::Create(C_LayerName_Radar, UIModule_Radar::GetManialink());
	Layers::SetType(C_LayerName_Radar, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_Radar);

	Layers::SetType(C_LayerName_Flash, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_Flash);

	Layers::Create(C_LayerName_Podium, UIModule_PodiumBackground::GetManialink());
	Layers::SetType(C_LayerName_Podium, CUILayer::EUILayerType::ScreenIn3d);
	Layers::Get(C_LayerName_Podium).AttachId = C_AttachId_Podium;
	// Don't attach yet; only explicitly during podium

	Layers::Create(C_LayerName_Spectator, UIModule_SpectatorControls::GetManialink());
	Layers::SetType(C_LayerName_Spectator, CUILayer::EUILayerType::Normal);
	Layers::Attach(C_LayerName_Spectator);

/**
 * Sends relevant progression mode settings values to the clients via netwrite.
 */
Void Net_SendModeScriptProgressionSettings(Integer NbMapsToWinMatch, Integer NbRoundsToWinMap, Integer NbFlagsToWinRound) {
	declare netwrite Integer Net_FlagRush_NbMapsToWinMatch for Teams[0];
	Net_FlagRush_NbMapsToWinMatch = NbMapsToWinMatch;

	declare netwrite Integer Net_FlagRush_NbRoundsToWinMap for Teams[0];
	Net_FlagRush_NbRoundsToWinMap = NbRoundsToWinMap;

	declare netwrite Integer Net_FlagRush_NbFlagsToWinRound for Teams[0];
	Net_FlagRush_NbFlagsToWinRound = NbFlagsToWinRound;
}

/* Lifecycle */

Void BeforeYield() {
	foreach (Event in Playloop::GetPendingEvents()) {
		switch (Event.Type) {
			case Playloop::C_EventType_FlagReset: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerPickedUpFlag: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerPassedFlag: {
				UIModule_FlagMarker::Update();
			}
			case Playloop::C_EventType_PlayerDroppedFlag: {
				UIModule_FlagMarker::Update();
			}
		}
	}
}

Void Yield() {
	declare Boolean UpdateRadar = False;
	foreach (Event in PlayerLifecycleEvents::GetPendingEvents()) {
		switch (Event.Type) {
			case PlayerLifecycleEvents::C_EventType_PlayerAdded: {
				if (Players.existskey(Event.Player.PlayerId)) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerRemoved: {
				if (!Event.Player.Previously.WasSpectator) {
					UpdateRadar = True;
				}
			}
			case PlayerLifecycleEvents::C_EventType_PlayerToSpectator, PlayerLifecycleEvents::C_EventType_SpectatorToPlayer: {
				UpdateRadar = True;
			}
		}
	}

	if (UpdateRadar) {
		Layers::Update(C_LayerName_Radar, UIModule_Radar::GetManialink());
	}
}

Void Unload() {
	Private_DestroyModuleLayers();
	Private_UnloadModules();
}

Void Load() {
	Unload();
	Private_LoadModules();
}

Void LoadHud() {
	Private_CreateModuleLayers();
}
