/**
 * A rather random collection of useful or convenience functions.
 *
 * #RequireContext CSmMode
 */

/**
 * Gets the score for a user with given login.
 * Returns null if there is no score for the given login.
 */
 CSmScore GetScore(Text _Login) {
	foreach(Score in Scores) {
		if(Score.User.Login == _Login) {
			return Score;
		}
	}
	return Null;
}

/**
 * Gets the user for a given login.
 * Returns null if there is no user for the given login.
 */
CUser GetUser(Text _Login) {
	foreach (User in Users) {
		if(User.Login == _Login) {
			return User;
		}
	}
	return Null;
}
