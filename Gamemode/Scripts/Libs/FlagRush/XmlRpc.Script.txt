// MARK: Structs

// Components

#Struct K_Player {
  Text Login;
  Text WebServicesUserId;
  Text Name;
  Integer Clan;
  Vec3 Position;
}

#Struct K_Landmark {
  Integer Id;
  Vec3 Position;
}

// Events

#Struct K_Event_Player_RequestedRespawn {
  Integer Timestamp;
  K_Player Player;
  Boolean Success;
}

#Struct K_Event_Player_Spawned {
  Integer Timestamp;
  K_Player Player;
  K_Landmark Landmark;
}

#Struct K_Event_Player_OutOfBounds {
  Integer Timestamp;
  K_Player Player;
}

#Struct K_Event_Player_Died {
  Integer Timestamp;
  K_Player Player;
}

#Struct K_Event_Flag_Reset {
  Integer Timestamp;
  K_Landmark FlagSpawn;
}

#Struct K_Event_Flag_PickedUp {
  Integer Timestamp;
  K_Player Carrier;
}

#Struct K_Event_Flag_Passed {
  Integer Timestamp;
  K_Player NewCarrier;
  K_Player PreviousCarrier;
}

#Struct K_Event_Flag_Dropped {
  Integer Timestamp;
  Vec3 Position;
  K_Player PreviousCarrier;
}

#Struct K_Event_Flag_CarrierTeleported {
  Integer Timestamp;
  K_Player Carrier;
}

#Struct K_Event_Flag_Scored {
  Integer Timestamp;
  K_Player Scorer;
  K_Player Assist;
}

// MARK: Constants

#Const C_Position_Invalid <-1., -1., -1.>
#Const C_Id_Invalid -1

#Const C_EventType_Player_RequestedRespawn    "FlagRush.Player.RequestedRespawn"
#Const C_EventType_Player_Spawned             "FlagRush.Player.Spawned"
#Const C_EventType_Player_OutOfBounds         "FlagRush.Player.OutOfBounds"
#Const C_EventType_Player_Died                "FlagRush.Player.Died"
#Const C_EventType_Flag_Reset                 "FlagRush.Flag.Reset"
#Const C_EventType_Flag_PickedUp              "FlagRush.Flag.PickedUp"
#Const C_EventType_Flag_Passed                "FlagRush.Flag.Passed"
#Const C_EventType_Flag_Dropped               "FlagRush.Flag.Dropped"
#Const C_EventType_Flag_OnCarrierTeleported   "FlagRush.Flag.CarrierTeleported"
#Const C_EventType_Flag_Scored                "FlagRush.Flag.Scored"

#Const C_Player_Invalid K_Player{
  Name = "Invalid Player",
  Position = <-1., -1., -1.>
}
#Const C_Landmark_Invalid K_Landmark{
  Id = -1,
  Position = <-1., -1., -1.>
}

// MARK: Helper Functions

Integer Private_IdToInteger(Ident _Id) {
  if (_Id == NullId) {
    return C_Id_Invalid;
  }
  return TL::ToInteger(TL::Replace("" ^ _Id, "#", ""));
}

K_Player Private_CreatePlayerStruct(CSmPlayer _Player) {
  if (_Player == Null) {
    return C_Player_Invalid;
  }
  return K_Player{
    Login = _Player.User.Login,
    WebServicesUserId = _Player.User.WebServicesUserId,
    Name = _Player.User.Name,
    Clan = _Player.CurrentClan,
    Position = _Player.Position
  };
}

K_Player Private_CreatePlayerStruct(CSmScore _Score) {
  if (_Score == Null) {
    return C_Player_Invalid;
  }

  // Try to find player for the score
  foreach (Player in Players) {
    if (Player.Score == _Score) {
      return Private_CreatePlayerStruct(Player);
    }
  }

  return K_Player{
    Login = _Score.User.Login,
    WebServicesUserId = _Score.User.WebServicesUserId,
    Name = _Score.User.Name,
    Clan = _Score.TeamNum,
    Position = C_Position_Invalid
  };
}

K_Landmark Private_CreateLandmarkStruct(CMapLandmark _Landmark) {
  if (_Landmark == Null) {
    return C_Landmark_Invalid;
  }
  return K_Landmark{
    Id = Private_IdToInteger(_Landmark.Id),
    Position = _Landmark.Position
  };
}

Void Private_SendCallback(Text _EventType, Text _EventDataJson) {
  XmlRpc::SendCallback(_EventType, _EventDataJson);
  Log::Debug("""XmlRpc: Sent callback "{{{ _EventType }}}" with data: {{{ _EventDataJson }}}""");
}

// MARK: Event Handlers

Void Private_OnPlayerRequestsRespawnButIsTooFast(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Player_RequestedRespawn Event = K_Event_Player_RequestedRespawn{
    Timestamp = Now,
    Player = Private_CreatePlayerStruct(Player),
    Success = False
  };
  Private_SendCallback(C_EventType_Player_RequestedRespawn, Event.tojson());
}

Void Private_OnPlayerRespawned(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Player_RequestedRespawn Event = K_Event_Player_RequestedRespawn{
    Timestamp = Now,
    Player = Private_CreatePlayerStruct(Player),
    Success = True
  };
  Private_SendCallback(C_EventType_Player_RequestedRespawn, Event.tojson());
}

Void Private_OnPlayerSpawned(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Player_Spawned Event = K_Event_Player_Spawned{
    Timestamp = Now,
    Player = Private_CreatePlayerStruct(Player),
    Landmark = Private_CreateLandmarkStruct(_Event.Landmark)
  };
  Private_SendCallback(C_EventType_Player_Spawned, Event.tojson());
}

Void Private_OnPlayerOutOfBounds(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Player_OutOfBounds Event = K_Event_Player_OutOfBounds{
    Timestamp = Now,
    Player = Private_CreatePlayerStruct(Player)
  };
  Private_SendCallback(C_EventType_Player_OutOfBounds, Event.tojson());
}

Void Private_OnPlayerDied(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Player_Died Event = K_Event_Player_Died{
    Timestamp = Now,
    Player = Private_CreatePlayerStruct(Player)
  };
  Private_SendCallback(C_EventType_Player_Died, Event.tojson());
}

Void Private_OnFlagReset(Playloop::K_Event _Event) {
  declare K_Event_Flag_Reset Event = K_Event_Flag_Reset{
    Timestamp = Now,
    FlagSpawn = Private_CreateLandmarkStruct(_Event.Landmark)
  };
  Private_SendCallback(C_EventType_Flag_Reset, Event.tojson());
}

Void Private_OnFlagPickedUp(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Flag_PickedUp Event = K_Event_Flag_PickedUp{
    Timestamp = Now,
    Carrier = Private_CreatePlayerStruct(Player)
  };
  Private_SendCallback(C_EventType_Flag_PickedUp, Event.tojson());
}

Void Private_OnFlagPassed(Playloop::K_Event _Event) {
  declare CSmPlayer PreviousCarrier <=> Playloop::GetPlayer(_Event.Player);
  declare CSmPlayer NewCarrier <=> Playloop::GetPlayer(_Event.RelatedPlayer);
  declare K_Event_Flag_Passed Event = K_Event_Flag_Passed{
    Timestamp = Now,
    NewCarrier = Private_CreatePlayerStruct(NewCarrier),
    PreviousCarrier = Private_CreatePlayerStruct(PreviousCarrier)
  };
  Private_SendCallback(C_EventType_Flag_Passed, Event.tojson());
}

Void Private_OnFlagDropped(Playloop::K_Event _Event) {
  // Do not use player. Might be disconnected
  declare CSmScore PreviousCarrierScore <=> Playloop::GetScore(_Event.Player);
  declare K_Event_Flag_Dropped Event = K_Event_Flag_Dropped{
    Timestamp = Now,
    Position = _Event.Position,
    PreviousCarrier = Private_CreatePlayerStruct(PreviousCarrierScore)
  };
  Private_SendCallback(C_EventType_Flag_Dropped, Event.tojson());
}

Void Private_OnFlagCarrierTeleported(Playloop::K_Event _Event) {
  declare CSmPlayer Player <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Flag_CarrierTeleported Event = K_Event_Flag_CarrierTeleported{
    Timestamp = Now,
    Carrier = Private_CreatePlayerStruct(Player)
  };
  Private_SendCallback(C_EventType_Flag_OnCarrierTeleported, Event.tojson());
}

Void Private_OnFlagScored(Playloop::K_Event _Event) {
  declare CSmPlayer Scorer <=> Playloop::GetPlayer(_Event.Player);
  declare K_Event_Flag_Scored Event = K_Event_Flag_Scored{
    Timestamp = Now,
    Scorer = Private_CreatePlayerStruct(Scorer)
  };

  // Do not use player. Might be disconnected
  declare CSmScore AssistScore <=> Playloop::GetScore(_Event.RelatedPlayer);
  if (AssistScore == Null) {
    Event.Assist = C_Player_Invalid;
  } else {
    Event.Assist = Private_CreatePlayerStruct(AssistScore);
  }
  Private_SendCallback(C_EventType_Flag_Scored, Event.tojson());
}

// MARK: Lifecycle

Void BeforeYield() {
  foreach (Event in Playloop::GetPendingEvents()) {
    switch (Event.Type) {
      case Playloop::C_EventType_PlayerRequestedRespawnButIsTooFast: Private_OnPlayerRequestsRespawnButIsTooFast(Event);
      case Playloop::C_EventType_PlayerRespawned: Private_OnPlayerRespawned(Event);
      case Playloop::C_EventType_PlayerSpawned: Private_OnPlayerSpawned(Event);
      case Playloop::C_EventType_PlayerOutOfBounds: Private_OnPlayerOutOfBounds(Event);
      case Playloop::C_EventType_PlayerDied: Private_OnPlayerDied(Event);
      case Playloop::C_EventType_FlagReset: Private_OnFlagReset(Event);
      case Playloop::C_EventType_PlayerPickedUpFlag: Private_OnFlagPickedUp(Event);
      case Playloop::C_EventType_PlayerPassedFlag: Private_OnFlagPassed(Event);
      case Playloop::C_EventType_PlayerDroppedFlag: Private_OnFlagDropped(Event);
      case Playloop::C_EventType_FlagCarrierTeleported: Private_OnFlagCarrierTeleported(Event);
      case Playloop::C_EventType_PlayerScoredFlag: Private_OnFlagScored(Event);
    }
  }

  foreach (Event in Progression::GetPendingEvents()) {
    switch (Event.Type) {
      case Progression::C_EventType_SegmentStart: {
        switch (Event.Segment) {
          case Progression::C_Segment_Server: {}
          case Progression::C_Segment_Match: {}
          case Progression::C_Segment_Map: {}
          case Progression::C_Segment_Round: {}
          case Progression::C_Segment_Turn: {}
          case Progression::C_Segment_Overtime: {}
          case Progression::C_Segment_WarmUp: {}
          case Progression::C_Segment_Pause: {}
          case Progression::C_Segment_Podium: {}
        }
      }
      case Progression::C_EventType_SegmentEnd: {
        switch (Event.Segment) {
          case Progression::C_Segment_Server: {}
          case Progression::C_Segment_Match: {}
          case Progression::C_Segment_Map: {}
          case Progression::C_Segment_Round: {}
          case Progression::C_Segment_Turn: {}
          case Progression::C_Segment_Overtime: {}
          case Progression::C_Segment_WarmUp: {}
          case Progression::C_Segment_Pause: {}
          case Progression::C_Segment_Podium: {}
        }
      }
    }
  }
}
