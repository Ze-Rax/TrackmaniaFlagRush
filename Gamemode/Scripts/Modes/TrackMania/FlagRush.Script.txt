  if (!Private_RoundIsTied()) {
    MB_StopRound();

  MB_StopRound();
} else if (Private_RoundTimeLimitReached()) {
  if (Private_RoundIsTied() && S_UseOvertime) {
    Progression::StartOvertime();
    FlagRush_XmlRpc::Send_Overtime();
  } else {
    Progression::SignalTimeLimitReached();
    MB_StopRound();
  }
}

	MB_StopTurn();

{ +++StartWarmUp+++ }
while(!WarmUpReadyUp::WarmUpShouldStop()) {
	MB_Yield();
	WarmUpReadyUp::Update();
	foreach (WuruEvent in WarmUpReadyUp::GetPendingEvents()) {
		switch (WuruEvent.Type) {
			case WarmUpReadyUp::C_Event_Type_TimeoutStart,
					WarmUpReadyUp::C_Event_Type_TimeoutUpdate,
					WarmUpReadyUp::C_Event_Type_TimeoutAbort: {
				// Refrech EndTime net variable when it's updated by WURU
				Progression::SetEndTime(EndTime);
			}
		}
	}
	+++WarmUpLoop+++
}
{ +++EndWarmUp+++ }

Progression::SetEndTime(-1);
Progression::EndWarmUp();
FlagRush_XmlRpc::Send_WarmUpEnd();
UIManager.UIAll.UISequence = PreviousSequence;

***Private_Yield_HandlePlayerLifecycleEvents***
***
declare Boolean ReinitTeamNames = False;
declare Boolean LastPlayerLeft = False;
foreach (Event in PlayerLifecycleEvents::GetPendingEvents()) {
	// Check if player event requires team reinitialization
	declare Boolean PlayerAdded = Event.Type == PlayerLifecycleEvents::C_EventType_PlayerAdded && Players.existskey(Event.Player.PlayerId);
	declare Boolean PlayerRemoved = Event.Type == PlayerLifecycleEvents::C_EventType_PlayerRemoved && !Event.Player.Previously.WasSpectator;
	declare Boolean PlayerClanChange = Event.Type == PlayerLifecycleEvents::C_EventType_PlayerClanChange;
	declare Boolean PlayerSpectatorChange = Event.Type == PlayerLifecycleEvents::C_EventType_PlayerToSpectator || Event.Type == PlayerLifecycleEvents::C_EventType_SpectatorToPlayer;
	ReinitTeamNames = ReinitTeamNames || PlayerAdded || PlayerRemoved || PlayerClanChange || PlayerSpectatorChange;

	// Skip match if no player is left after player left or moved to spectator
	LastPlayerLeft = LastPlayerLeft || Players.count == 0 && (PlayerRemoved || Event.Type == PlayerLifecycleEvents::C_EventType_PlayerToSpectator);
}

if (ReinitTeamNames) {
	Teams::Init();
}

// Skip running match if the last player left, but not if we're before the first round of the first map
declare Boolean MatchWasStarted =  MB_GetMapCount() > 1 || MB_GetRoundCount() > 0; // At least first round on first map was started
if (S_SkipMatchOnLastPlayerLeave && LastPlayerLeft && MatchWasStarted && Map != Null) {
	Progression::SignalMatchSkippedDueToNoPlayersLeft();
	Log::Info("Current match was skipped as there are no players remaining");
	Progression::InvalidateMatch();
	Progression::RequestMatchEnd(); // Keep for consistency
	MB_StopMatch();
	if (Progression::WarmUpIsRunning()) {
		WarmUpReadyUp::Stop();
	}
}

***Private_Yield_HandleCriticalModeScriptSettingsUpdates***
***
declare Boolean ReinitTeamsConfig = False;
declare Boolean UpdateVehicleTunings = False;
declare Boolean RestartTurn = False;
foreach(Event in Settings::GetPendingEvents()) {
	switch (Event.Name) {
		// Team Configs
		case SettingNames::S_UseClubTags, SettingNames::S_Team1Name, SettingNames::S_Team1Hue, SettingNames::S_Team2Name, SettingNames::S_Team2Hue: {
			ReinitTeamsConfig = True;
		}
		// Collision
		case SettingNames::S_UseCollisions: {
			UsePvPCollisions = S_UseCollisions;
			RestartTurn = True; // Respawn all players; this is easier
		}
		// Vehicle Tunings
		case SettingNames::S_TuningStadiumAcceleration, SettingNames::S_TuningStadiumAdherence, SettingNames::S_TuningStadiumControl,
					SettingNames::S_TuningSnowAcceleration, SettingNames::S_TuningSnowAdherence, SettingNames::S_TuningSnowControl,
					SettingNames::S_TuningRallyAcceleration, SettingNames::S_TuningRallyAdherence, SettingNames::S_TuningRallyControl,
					SettingNames::S_TuningDesertAcceleration, SettingNames::S_TuningDesertAdherence, SettingNames::S_TuningDesertControl: {
			UpdateVehicleTunings = True;
			RestartTurn = True; // Could just apply vehicle tunings, but this is easier
		}
		// Forced Vehicles
		case SettingNames::S_ForceEnabledVehiclesCsv: {
			declare Vehicles::K_Vehicle[] ForcedEnabledVehicles;
			foreach (CsvEntry in TL::Split(",", Settings::GetText(SettingNames::S_ForceEnabledVehiclesCsv))) {
				declare Text VehicleId = TL::ToLowerCase(TL::Trim(CsvEntry));
				if (Vehicles::IsValid(VehicleId)) {
					ForcedEnabledVehicles.add(Vehicles::Get(VehicleId));
				}
			}
			EnabledVehicles::Mode_ForceEnabledVehicles(ForcedEnabledVehicles);
			RestartTurn = True; // Respawn players with disallowed vehicles; this is easier
		}
		// Physics simulation
		case SettingNames::S_TrustClientSimu, SettingNames::S_UseCrudeExtrapolation: {
			RestartTurn = True; // Respawn all players; this is easier
		}
		// Debug
		case SettingNames::S_Debug: {
			if (S_Debug) {
				Log::SetLevel(Log::C_Level_Debug);
			} else {
				Log::SetLevel(Log::C_Level_Info);
			}
		}
	}
}

if (ReinitTeamsConfig) {
	Teams::UseClubTags(S_UseClubTags);
	Teams::SetTeam1NameOverride(S_Team1Name);
	Teams::SetTeam2NameOverride(S_Team2Name);
	Teams::SetTeam1HueOverride(S_Team1Hue);
	Teams::SetTeam2HueOverride(S_Team2Hue);
	Teams::Init();
}

if (UpdateVehicleTunings) {
	declare Vehicles::K_Tuning StadiumTuning = Vehicles::C_Tuning_Default;
	StadiumTuning.Acceleration =	S_TuningStadiumAcceleration;
	StadiumTuning.Adherence =			S_TuningStadiumAdherence;
	StadiumTuning.Control =				S_TuningStadiumControl;
	Vehicles::SetTuning(C_Vehicle_Stadium.Id, StadiumTuning);

	declare Vehicles::K_Tuning SnowTuning = Vehicles::C_Tuning_Default;
	SnowTuning.Acceleration =			S_TuningSnowAcceleration;
	SnowTuning.Adherence =				S_TuningSnowAdherence;
	SnowTuning.Control =					S_TuningSnowControl;
	Vehicles::SetTuning(C_Vehicle_Snow.Id, SnowTuning);

	declare Vehicles::K_Tuning RallyTuning = Vehicles::C_Tuning_Default;
	RallyTuning.Acceleration =		S_TuningRallyAcceleration;
	RallyTuning.Adherence =				S_TuningRallyAdherence;
	RallyTuning.Control =					S_TuningRallyControl;
	Vehicles::SetTuning(C_Vehicle_Rally.Id, RallyTuning);

	declare Vehicles::K_Tuning DesertTuning = Vehicles::C_Tuning_Default;
	DesertTuning.Acceleration =		S_TuningDesertAcceleration;
	DesertTuning.Adherence =			S_TuningDesertAdherence;
	DesertTuning.Control =				S_TuningDesertControl;
	Vehicles::SetTuning(C_Vehicle_Desert.Id, DesertTuning);
}

if (RestartTurn && MB_TurnIsRunning()) {
	Progression::SignalRestartTurnDueToCriticalModeSettingChanged();
	MB_StopTurn();
}

***Private_Yield_HandleProgressionsCommands***
***
foreach (Command in Commands::GetPendingEvents()) {
	// Display a message in chat
	declare Text CommandText;

	// Handle the command
	switch(Command.Name) {
		case CommandNames::Command_EndWarmup: {
			WarmUpReadyUp::Stop();
			CommandText = "The warmup was ended.";
		}
		case CommandNames::Command_PauseRound: {
			Progression::RequestRoundPause();
			CommandText = "The round was paused.";
		}
		case CommandNames::Command_SkipRound: {
			Progression::InvalidateRound();
			Progression::RequestRoundEnd();
			CommandText = "The round was skipped.";
		}
		case CommandNames::Command_SkipMap: {
			Progression::InvalidateMap();
			Progression::RequestMapEnd();
			CommandText = "The map was skipped.";
		}
		case CommandNames::Command_SetTeam1RoundScore: {
			declare Integer Value = ML::Max(Command.ValueInteger, 0);
			Scores::SetClanRoundPoints(1, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[0].Name }}} round score to {{{ Value }}}.""";
		}
		case CommandNames::Command_SetTeam2RoundScore: {
			declare Integer Value = ML::Max(Command.ValueInteger, 0);
			Scores::SetClanRoundPoints(2, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[1].Name }}} round score to {{{ Value }}}.""";
		}
		case CommandNames::Command_SetTeam1MapScore: {
			declare Integer Value = ML::Clamp(Command.ValueInteger, 0, S_NbRoundsToWinMap);
			Scores::SetClanMapPoints(1, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[0].Name }}} map score to {{{ Value }}}.""";
		}
		case CommandNames::Command_SetTeam2MapScore: {
			declare Integer Value = ML::Clamp(Command.ValueInteger, 0, S_NbRoundsToWinMap);
			Scores::SetClanMapPoints(2, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[1].Name }}} map score to {{{ Value }}}.""";
		}
		case CommandNames::Command_SetTeam1MatchScore: {
			declare Integer Value = ML::Clamp(Command.ValueInteger, 0, S_NbMapsToWinMatch);
			Scores::SetClanMatchPoints(1, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[0].Name }}} match score to {{{ Value }}}.""";
		}
		case CommandNames::Command_SetTeam2MatchScore: {
			declare Integer Value = ML::Clamp(Command.ValueInteger, 0, S_NbMapsToWinMatch);
			Scores::SetClanMatchPoints(2, Value);
			Scores::Net_SendTeamScores();
			CommandText = """Set {{{ Teams[1].Name }}} match score to {{{ Value }}}.""";
		}
		case CommandNames::Command_ClearScores: {
			Scores::Clear();
			CommandText = "Cleared all scores.";
		}
		case CommandNames::Command_BalanceTeams: {
			ServerAdmin.AutoTeamBalance();
			CommandText = "The teams were balanced.";
		}
		case CommandNames::Command_SetBots: {
			declare Integer Amount = ML::Max(Command.ValueInteger, 0);
			Users_SetNbFakeUsers(Amount, Amount);
			CommandText = """Set {{{ Amount }}} bots.""";
		}
		case CommandNames::Command_SetTimeLeft: {
			if (!MB_TurnIsRunning()) {
				break;
			}
			Progression::SetEndTime(Now + Command.ValueInteger * 1000);
			CommandText = """Set time limit to {{{ Command.ValueInteger }}} seconds remaining.""";
		}
		case CommandNames::Command_ApproveWarmUpEnd: {
			WarmUpReadyUp::SetWarmUpEndApproval(Command.ValueBoolean);
			Progression::SignalWarmUpApprovalChanged();
			CommandText = """Approvel given: {{{ Command.ValueBoolean }}}.""";
		}
		case CommandNames::Command_Announce: {
			UIManager.UIAll.QueueMessage(3000, 999, CUIConfig::EMessageDisplay::Big, Command.ValueText, CUIConfig::EUISound::Default, 1);
			CommandText = """Message: $<{{{ Command.ValueText }}}$>""";
		}
		default: return;
	}

	declare Text UnprefixedCommandName = Command::RemovePrefix(Command.Name);
	declare Text Message = """Received command "{{{ UnprefixedCommandName }}}" from {{{ Command.Source.Name }}}.""";
	if (CommandText != "") {
		Message ^= " " ^ CommandText;
	}
	Chat::Send(Message);
}

Boolean Private_MatchPointLimitReached() {
	return ML::Max(Scores::GetClanMatchPoints(1), Scores::GetClanMatchPoints(2)) >= S_NbMapsToWinMatch;
}

Boolean Private_MapPointLimitReached() {
	return ML::Max(Scores::GetClanMapPoints(1), Scores::GetClanMapPoints(2)) >= S_NbRoundsToWinMap;
}

Boolean Private_RoundPointLimitReached() {
	if (S_NbFlagsToWinRound <= 0) {
		return False;
	}
	declare Integer RoundPointsTeam1 = Scores::GetClanRoundPoints(1);
	declare Integer RoundPointsTeam2 = Scores::GetClanRoundPoints(2);
	return ML::Max(RoundPointsTeam1, RoundPointsTeam2) >= S_NbFlagsToWinRound;
}

Boolean Private_RoundTimeLimitReached() {
	if (S_RoundTimeLimitSeconds <= 0) {
		return False;
	}
	return Now > EndTime;
}

Boolean Private_RoundIsTied() {
	return Scores::EvaluateRound() == 0;
}
